def sub(path, pairs):
    s = open(path, encoding="utf-8", newline="").read()
    crlf = "\r\n" in s
    s = s.replace("\r\n", "\n")
    for a, b in pairs:
        assert s.count(a) == 1, (path, s.count(a), a[:90])
        s = s.replace(a, b)
    return path, s.replace("\n", "\r\n") if crlf else s

    # EP-LAPTOP floor 2026-11-24: 374 m/s (full delay +-0.33 ms) with 0.16 ms scatter from step to step
