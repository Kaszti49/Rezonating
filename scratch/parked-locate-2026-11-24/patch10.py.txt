def sub(path, pairs):
    s = open(path, encoding="utf-8", newline="").read()
    crlf = "\r\n" in s
    s = s.replace("\r\n", "\n")
    for a, b in pairs:
        assert s.count(a) == 1, (path, s.count(a), a[:110])
        s = s.replace(a, b)
    OUT[path] = s.replace("\n", "\r\n") if crlf else s
