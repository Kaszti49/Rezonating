def tukey(n,a=.1):
    w=np.ones(n); k=int(a*(n-1)/2); w[:k+1]=.5*(1+np.cos(np.pi*(2*np.arange(k+1)/(a*(n-1))-1))); w[-k-1:]=w[:k+1][::-1]; return w
