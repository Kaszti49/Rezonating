def feats(X,f,o,i,kind):
    w=win(X,o,i)*np.hanning(700)[:,None]; s=np.fft.rfft(w,2048,axis=0); fr=np.fft.rfftfreq(2048,1/f); v=[]

        k=(fr>=lo)&(fr<hi); c=(s[k,1]*np.conj(s[k,0])).sum(); a=np.sqrt((abs(s[k,0])**2).sum()*(abs(s[k,1])**2).sum()); v+= [np.log((abs(s[k,0])**2).sum()/(abs(s[k,1])**2).sum()), (c/a).real, (c/a).imag]
    if kind=="raw":
        x=win(X,o,i,20,180); x=x/np.abs(x).max(); return np.r_[np.array(v), (x[:,0][::4]),(x[:,1][::4])]
    return np.array(v)

def nearest(tr,trl,te):
    mu,sd=tr.mean(0),tr.std(0)+1e-9; cl=sorted(set(trl)); cen={c:((tr[np.array(trl)==c]-mu)/sd).mean(0) for c in cl}
    return [min(cl,key=lambda c:np.sum(((t-mu)/sd-cen[c])**2)) for t in te]
