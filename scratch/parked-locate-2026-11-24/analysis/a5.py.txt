def rob(v): return 1.4826*np.median(abs(v-np.median(v)))
# LS estimator: rho = a h + b dh/dt, band limited
def ls(i,pre,post,lo,hi):
    w=X[i-pre:i+post].astype(float); w=w-np.median(w,0); w=w*np.hanning(len(w))[:,None]; n=2048
    S=np.fft.rfft(w,n,axis=0); f=np.fft.rfftfreq(n,1/fs); m=(f>=lo)&(f<=hi); x0,x1=S[m,0],-S[m,1]; h=(x0+x1)/2; r=(x0-x1)/2; D=2j*np.pi*f[m]*h
    A=np.c_[np.r_[h.real,h.imag],np.r_[D.real,D.imag]]; y=np.r_[r.real,r.imag]; sol=np.linalg.lstsq(A,y,rcond=None)[0]
    return sol[1]*2e3, sol[0]
