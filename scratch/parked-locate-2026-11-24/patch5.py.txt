def load(path):
    s = open(path, encoding="utf-8", newline="").read()
    return s.replace("\r\n", "\n"), "\r\n" in s


def sub(path, pairs):
    s, crlf = load(path)
    for a, b in pairs:
        assert s.count(a) == 1, (path, s.count(a), a[:100])
        s = s.replace(a, b)
    OUT[path] = s.replace("\n", "\r\n") if crlf else s


def between(path, start, end, new, pairs=()):
    s, crlf = load(path)
    for a, b in pairs:
        assert s.count(a) == 1, (path, s.count(a), a[:100])
        s = s.replace(a, b)
    i = s.index(start)
    j = s.index(end, i) if end else len(s)
    s = s[:i] + new + s[j:]
    OUT[path] = s.replace("\n", "\r\n") if crlf else s
